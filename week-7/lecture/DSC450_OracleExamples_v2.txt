SELECT LEVEL n
    FROM dual
    CONNECT BY LEVEL <= 20;

SELECT n
	FROM (SELECT LEVEL n FROM dual CONNECT BY LEVEL <= 100)
	WHERE n > 1
	AND NOT EXISTS (
	  SELECT 1
	  FROM (SELECT LEVEL d FROM dual CONNECT BY LEVEL <= 100)
	  WHERE d BETWEEN 2 AND FLOOR(SQRT(n))
	  AND MOD(n, d) = 0
	)
	ORDER BY n;

CREATE TABLE departments(
    department_id NUMBER PRIMARY KEY,

    department_id NUMBER,
    salary NUMBER,
    CONSTRAINT fk_dept FOREIGN KEY (department_id)

    emplyee_id NUMBER,
    salary NUMBER,
    audit_date DATE DEFAULT SYSDATE
);

WITH employees_fin AS 
( SELECT employee_id, first_name, department_id 
  FROM employees WHERE department_id = 10 ) 
SELECT first_name FROM employees_fin 
ORDER BY first_name FETCH FIRST 5 ROWS ONLY;


WITH dept_sums AS (
  SELECT department_id, SUM(salary) AS total_salary
  FROM employees
  GROUP BY department_id
),
high_salary AS (
  SELECT department_id
  FROM dept_sums
  WHERE total_salary > 200000
)
SELECT d.department_name, h.department_id, ds.total_salary
FROM departments d
JOIN high_salary h ON d.department_id = h.department_id
JOIN dept_sums ds ON d.department_id = ds.department_id;


WITH org_chart (employee_id, first_name, manager_id, lvl) AS (
  SELECT employee_id, first_name, manager_id, 1 AS lvl
  FROM employees
  WHERE manager_id IS NULL
  UNION ALL
  SELECT e.employee_id, e.first_name, e.manager_id, o.lvl + 1
  FROM employees e
  JOIN org_chart o ON e.manager_id = o.employee_id
)
SELECT LPAD(' ', (lvl - 1) * 2) || first_name AS hierarchy,
       lvl
FROM org_chart
ORDER BY lvl, first_name;

-- Running total per service
SELECT
  service,
  day,
  value,
  SUM(value) OVER (
    PARTITION BY service
    ORDER BY day
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
  ) AS running_total
FROM metrics
ORDER BY service, day;

-- Day-over-day change per service
SELECT
  service,
  day,
  value,
  (value - LAG(value) OVER (PARTITION BY service ORDER BY day)) AS delta
FROM metrics
ORDER BY service, day;

-- Top day per service
SELECT service, day, value
FROM (
  SELECT
    service,
    day,
    value,
    ROW_NUMBER() OVER (
      PARTITION BY service
      ORDER BY value DESC, day DESC
    ) AS rn
  FROM metrics
)
WHERE rn = 1
ORDER BY service;
